## Design
This program had a three layer split (packets, state machines, I/O) which was meant to make our protocol logic be separate and testable, independent of the OS until required. The packets served as pure byte level function meant to handle the simple packet data, whether to encode, checksum, or validate. The second layer, state machines, were meant to fully run through the logical side in a testable way, which means no OS integration such as sockets, clocks or files. The final layer, I/O, is meant to finalize the actual interface to the OS and provide sockets and access to complete the entire cycle of each protocol.

| 1      | 0     |   103.546 |   103.530 |   103.537 |  103.538 |               9.9 |

| 1      | 0.05  |   133.865 |   133.339 |   132.109 |  133.104 |               7.7 |
| 16     | 0.05  |    27.079 |    24.362 |    23.205 |   24.882 |              41.2 |

**(a) Round-Trip Time.** With window 1 every packet
waits a full round trip before the next is sent so the 1024 DATA packets plus the FIN
equal a total of 1025 packets. 103.538 s / 1025 = 101.0 ms per packet. The relay was
configured to add exactly 100 ms , so the extra 1 ms is the
one-time HELLO/OK handshake, which costs one extra round trip at the start. Packet processing on both ends and OS work are all tiny contributions that also add up to the additional time.

**(b) Window Speedup Time.** Measured as 103.538 / 6.604 =
15.7, close, but not exactly the 16 factor. On a lossless channel a window-16 sender
pipelines up to 16 packets per, so the data phase takes around 1024/16 = 64 round trips
instead of 1024. The shortfall from 16× is fixed overhead that does not shrink
with window size like the HELLO/OK handshake at the start and the FIN/ACK exchange at
the end each cost a full round trip regardless of window. With only 65 trips in total,
those constants give some efficiency reduction to the ratio with small OS jitter
on top.

**(c) Proportional Loss.** Loss
slows window 1 by 29% (103.5 → 133.1 s) but window 16 by 277% (6.6 → 24.9 s).
This is the Go-Back-N trade-off in action. With window 1, a lost packet/ACK
costs one timeout and a retransmission of exactly one packet so the recovery is only 1 round trip
and nothing that already arrived is resent. With window 16, the same single loss
triggers a 250 ms timeout and a full retransmission of up to 16 packets, most of
which the receiver already has and will throw out.

#define HANDSHAKE_ATTEMPTS 5

#define POLL_CAP_MS 60000

static long now_ms(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (long)ts.tv_sec * 1000L + ts.tv_nsec / 1000000L;

/* Resolve <relay> and open the one UDP socket used for the entire run. */
static int open_relay(const char *host, int port, struct sockaddr_storage *addr_out, socklen_t *len_out)
{
    char portstr[8];
    snprintf(portstr, sizeof(portstr), "%d", port);

    struct addrinfo hints;
    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_DGRAM;

    struct addrinfo *res = NULL;
    int err = getaddrinfo(host, portstr, &hints, &res);
    if (err != 0 || res == NULL) {
        fprintf(stderr, "error: cannot resolve relay '%s': %s\n", host,
                err != 0 ? gai_strerror(err) : "no results");
        return -1;
    }

    int fd = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
    if (fd < 0) {
        fprintf(stderr, "error: socket failed: %s\n", strerror(errno));
        freeaddrinfo(res);
        return -1;
    }
    int fl = fcntl(fd, F_GETFL, 0);
    if (fl < 0 || fcntl(fd, F_SETFL, fl | O_NONBLOCK) < 0) {
        fprintf(stderr, "error: fcntl(O_NONBLOCK) failed: %s\n", strerror(errno));
        close(fd);
        freeaddrinfo(res);
        return -1;
    }
    memcpy(addr_out, res->ai_addr, res->ai_addrlen);
    *len_out = res->ai_addrlen;
    freeaddrinfo(res);
    return fd;
}

/* Registration handshake: send HELLO, wait up to 1 s for OK/ERR, resend on
 * silence, give up after HANDSHAKE_ATTEMPTS total attempts. */
static int handshake(int fd, const struct sockaddr_storage *addr, socklen_t alen,
                     const char *session, int is_send, double loss, double corrupt, double dup)
{
    char hello[96];
    if (is_send)
        snprintf(hello, sizeof(hello), "HELLO %s send %.6f %.6f %.6f", session, loss, corrupt, dup);
    else
        snprintf(hello, sizeof(hello), "HELLO %s recv", session);

    uint8_t reply[256];
    for (int attempt = 1; attempt <= HANDSHAKE_ATTEMPTS; attempt++) {
        if (sendto(fd, hello, strlen(hello), 0, (const struct sockaddr *)addr, alen) < 0) {
            fprintf(stderr, "error: sendto failed: %s\n", strerror(errno));
            return -1;
        }
        struct pollfd pfd;
        pfd.fd = fd;
        pfd.events = POLLIN;
        pfd.revents = 0;
        if (poll(&pfd, 1, HANDSHAKE_WAIT_MS) <= 0)
            continue;
        ssize_t n = recvfrom(fd, reply, sizeof(reply), 0, NULL, NULL);
        if (n <= 0)
            continue;
        if (memcmp(reply, "OK", 2) == 0)
            return 0;
        if (memcmp(reply, "ERR", 3) == 0) {
            fprintf(stderr, "error: relay refused: %.*s\n", (int)n, reply);
            return -1;
        }
    }
    fprintf(stderr, "error: no relay reply after %d attempts\n", HANDSHAKE_ATTEMPTS);
    return -1;
}

static int run_send(const config_t *cfg)
{

        fclose(f);
        return 2;

        fclose(f);
        return 2;

        pfd.revents = 0;
        if (poll(&pfd, 1, pt) < 0) {
            if (errno == EINTR)
                continue;
            fprintf(stderr, "error: poll failed: %s\n", strerror(errno));
            goto done;
        }
        for (;;) {
            ssize_t n = recvfrom(fd, rbuf, sizeof(rbuf), 0, NULL, NULL);
            if (n < 0) {
                if (errno == EINTR || errno == EAGAIN)
                    break;
                fprintf(stderr, "error: recvfrom failed: %s\n", strerror(errno));
                goto done;
            }
            if (n == 0)
                break;
            sx_on_packet(&st, now_ms(), rbuf, (size_t)n);
        }
        sx_tick(&st, now_ms());
    }

done:
    close(fd);
    free(file);
    return rc;
}

static int run_recv(const config_t *cfg)
{
    struct sockaddr_storage addr;
    socklen_t alen = 0;
    int fd = open_relay(cfg->relay, cfg->port, &addr, &alen);
    if (fd < 0)
        return 2;
    if (handshake(fd, &addr, alen, cfg->session, 0, 0.0, 0.0, 0.0) != 0) {
        close(fd);
        return 2;
    }

    FILE *f = fopen(cfg->file, "wb");
    if (f == NULL) {
        fprintf(stderr, "error: cannot open '%s' for writing: %s\n", cfg->file, strerror(errno));
        close(fd);
        return 2;
    }

    rx_state_t st;
    rx_init(&st, now_ms());

    uint8_t rbuf[IO_BUF_SIZE];
    uint8_t ackbuf[PKT_HEADER_LEN];
    int rc = 2;
    for (;;) {
        long now = now_ms();
        rx_action_t act = rx_tick(&st, now);
        if (act == RX_ACT_DONE) {
            rc = 0;
            break;
        }
        if (act == RX_ACT_GIVEUP) {
            fprintf(stderr, "error: no valid packet for %d ms\n", RX_IDLE_GIVEUP_MS);
            goto done;
        }

        long wait_ms = st.finished ? st.linger_until_ms - now : st.last_valid_ms + RX_IDLE_GIVEUP_MS - now;
        int pt = (int)(wait_ms > POLL_CAP_MS ? POLL_CAP_MS : (wait_ms < 1 ? 1 : wait_ms));
        struct pollfd pfd;
        pfd.fd = fd;
        pfd.events = POLLIN;
        pfd.revents = 0;
        if (poll(&pfd, 1, pt) < 0) {
            if (errno == EINTR)
                continue;
            fprintf(stderr, "error: poll failed: %s\n", strerror(errno));
            goto done;
        }
        for (;;) {
            ssize_t n = recvfrom(fd, rbuf, sizeof(rbuf), 0, NULL, NULL);
            if (n < 0) {
                if (errno == EINTR || errno == EAGAIN)
                    break;
                fprintf(stderr, "error: recvfrom failed: %s\n", strerror(errno));
                goto done;
            }
            if (n == 0)
                break;
            rx_action_t a = rx_on_packet(&st, now_ms(), rbuf, (size_t)n);
            if (a == RX_ACT_WRITE_ACK && fwrite(st.payload, 1, st.payload_len, f) != st.payload_len) {
                fprintf(stderr, "error: short write to '%s'\n", cfg->file);
                goto done;
            }
            if (a == RX_ACT_WRITE_ACK || a == RX_ACT_ACK) {
                int alen2 = pkt_encode(ackbuf, sizeof(ackbuf), PKT_TYPE_ACK, st.expected, NULL, 0);
                if (sendto(fd, ackbuf, (size_t)alen2, 0, (const struct sockaddr *)&addr, alen) < 0) {
                    fprintf(stderr, "error: sendto failed: %s\n", strerror(errno));
                    goto done;
                }
            }
        }
    }

done:
    fclose(f);
    close(fd);
    return rc;
}

int main(int argc, char **argv)
{
    config_t cfg;
    int code = parse_args(argc, argv, &cfg);
    if (code != 0)
        return code;
    if (argc == 1)
        return 0;

    if (cfg.mode == MODE_SEND)
        return run_send(&cfg);
    return run_recv(&cfg);
}

#include "pkt.h"
#include <string.h>

uint16_t pkt_checksum(const uint8_t *data, size_t len)
{
    uint32_t sum = 0;
    size_t i = 0;
    for (; i + 1 < len; i += 2)
        sum += ((uint32_t)data[i] << 8) | (uint32_t)data[i + 1];
    if (i < len)
        sum += (uint32_t)data[i] << 8;
    while (sum >> 16)
        sum = (sum & 0xffffu) + (sum >> 16);
    return (uint16_t)(~sum & 0xffffu);
}

int pkt_encode(uint8_t *buf, size_t cap, int type, uint32_t seq,
               const uint8_t *payload, uint16_t payload_len)
{
    if (type < PKT_TYPE_DATA || type > PKT_TYPE_FIN)
        return -1;
    if (payload_len > PKT_MAX_PAYLOAD)
        return -1;
    size_t total = (size_t)PKT_HEADER_LEN + payload_len;
    if (cap < total)
        return -1;

    buf[0] = (uint8_t)type;
    buf[1] = 0;
    buf[2] = 0;
    buf[3] = 0;
    buf[4] = (uint8_t)(seq >> 24);
    buf[5] = (uint8_t)(seq >> 16);
    buf[6] = (uint8_t)(seq >> 8);
    buf[7] = (uint8_t)seq;
    buf[8] = (uint8_t)(payload_len >> 8);
    buf[9] = (uint8_t)payload_len;
    if (payload_len != 0)
        memcpy(buf + PKT_HEADER_LEN, payload, payload_len);

    uint16_t csum = pkt_checksum(buf, total);
    buf[2] = (uint8_t)(csum >> 8);
    buf[3] = (uint8_t)csum;
    return (int)total;
}

int pkt_validate(const uint8_t *buf, size_t len, pkt_fields_t *out)
{
    if (len < PKT_HEADER_LEN)
        return -1;
    uint16_t length = (uint16_t)(((uint16_t)buf[8] << 8) | (uint16_t)buf[9]);
    if (length > PKT_MAX_PAYLOAD)
        return -1;
    if ((size_t)PKT_HEADER_LEN + length != len)
        return -1;
    int type = buf[0];
    if (type < PKT_TYPE_DATA || type > PKT_TYPE_FIN)
        return -1;
    if (buf[1] != 0)
        return -1;
    if (pkt_checksum(buf, len) != 0)
        return -1;

    out->type = type;
    out->seq = ((uint32_t)buf[4] << 24) | ((uint32_t)buf[5] << 16) |
               ((uint32_t)buf[6] << 8) | (uint32_t)buf[7];
    out->length = length;
    return 0;
}

#include "rx.h"

void rx_init(rx_state_t *st, long now_ms)
{
    st->expected = 0;
    st->finished = 0;
    st->linger_until_ms = 0;
    st->last_valid_ms = now_ms;
    st->payload = NULL;
    st->payload_len = 0;
}

rx_action_t rx_on_packet(rx_state_t *st, long now_ms, const uint8_t *buf, size_t len)
{
    pkt_fields_t f;
    if (pkt_validate(buf, len, &f) != 0)
        return RX_ACT_NONE;

    st->last_valid_ms = now_ms;

    if (st->finished)
        return RX_ACT_ACK;

    if (f.type == PKT_TYPE_DATA) {
        if (f.seq != st->expected)
            return RX_ACT_ACK;
        st->payload = buf + PKT_HEADER_LEN;
        st->payload_len = f.length;
        st->expected += 1;
        return RX_ACT_WRITE_ACK;
    }

    if (f.type == PKT_TYPE_FIN) {
        if (f.seq != st->expected)
            return RX_ACT_ACK;
        st->finished = 1;
        st->expected += 1;
        st->linger_until_ms = now_ms + RX_LINGER_MS;
        st->payload = NULL;
        st->payload_len = 0;
        return RX_ACT_ACK;
    }

    return RX_ACT_NONE;
}

rx_action_t rx_tick(rx_state_t *st, long now_ms)
{
    if (st->finished) {
        if (now_ms >= st->linger_until_ms)
            return RX_ACT_DONE;
        return RX_ACT_NONE;
    }
    if (now_ms - st->last_valid_ms >= RX_IDLE_GIVEUP_MS)
        return RX_ACT_GIVEUP;
    return RX_ACT_NONE;
}

#include <string.h>

#include "sx.h"

static void sx_timer_start(sx_state_t *st, long now_ms)
{
    if (!st->timer_running) {
        st->timer_running = 1;
        st->deadline_ms = now_ms + st->timeout_ms;
    }
}

static void sx_timer_restart(sx_state_t *st, long now_ms)
{
    st->timer_running = 1;
    st->deadline_ms = now_ms + st->timeout_ms;
}

void sx_init(sx_state_t *st, int window, long timeout_ms, const uint8_t *file, size_t file_len)
{
    memset(st, 0, sizeof(*st));
    st->window = window;
    st->timeout_ms = timeout_ms;
    st->file = file;
    st->file_len = file_len;
    st->n_data = (uint32_t)((file_len + PKT_MAX_PAYLOAD - 1u) / PKT_MAX_PAYLOAD);
}

void sx_on_packet(sx_state_t *st, long now_ms, const uint8_t *buf, size_t len)
{
    pkt_fields_t f;
    if (pkt_validate(buf, len, &f) != 0)
        return;
    if (f.type != PKT_TYPE_ACK)
        return;
    if (st->done || st->gave_up)
        return;

    if (st->fin_sent && !st->fin_acked) {
        if (f.seq == st->n_data + 1u) {
            st->fin_acked = 1;
            st->done = 1;
            st->timer_running = 0;
        }
        return;
    }

    if (f.seq > st->base) {
        uint32_t seq = f.seq;
        if (seq > st->n_data)
            seq = st->n_data;
        st->base = seq;
        if (st->next > st->base)
            sx_timer_restart(st, now_ms);
        else
            st->timer_running = 0;
    }
}

void sx_tick(sx_state_t *st, long now_ms)
{
    if (st->done || st->gave_up)
        return;
    if (!st->timer_running || now_ms < st->deadline_ms)
        return;

    if (st->base != st->last_base) {
        st->last_base = st->base;
        st->stuck_timeouts = 0;
    }
    st->stuck_timeouts += 1;
    if (st->stuck_timeouts >= SX_GIVEUP_TIMEOUTS) {
        st->gave_up = 1;
        st->timer_running = 0;
        return;
    }

    st->retransmit = 1;
    if (st->fin_sent && !st->fin_acked) {
        st->rt_fin = 1;
    } else {
        st->rt_fin = 0;
        st->rt_base = st->base;
        st->rt_next = st->next;
        st->rt_cursor = st->base;
    }
    sx_timer_restart(st, now_ms);
}

int sx_next_send(sx_state_t *st, long now_ms, sx_send_t *out)
{
    if (st->done || st->gave_up)
        return 0;

    if (st->retransmit) {
        if (st->rt_fin) {
            st->retransmit = 0;
            out->buf = st->fin_pkt.buf;
            out->len = st->fin_pkt.len;
            return 1;
        }
        if (st->rt_cursor < st->rt_next) {
            uint32_t seq = st->rt_cursor++;
            sx_pkt_t *p = &st->w[seq % SX_MAX_WINDOW];
            out->buf = p->buf;
            out->len = p->len;
            return 1;
        }
        st->retransmit = 0;
    }

    if (!st->fin_sent && st->base == st->n_data) {
        int n = pkt_encode(st->fin_pkt.buf, PKT_MAX_SIZE, PKT_TYPE_FIN, st->n_data, NULL, 0);
        st->fin_pkt.len = n;
        st->fin_sent = 1;
        sx_timer_start(st, now_ms);
        out->buf = st->fin_pkt.buf;
        out->len = n;
        return 1;
    }

    if (st->next < st->n_data && st->next < st->base + (uint32_t)st->window) {
        uint32_t seq = st->next;
        size_t off = (size_t)seq * (size_t)PKT_MAX_PAYLOAD;
        size_t remaining = st->file_len - off;
        uint16_t plen = (uint16_t)(remaining > PKT_MAX_PAYLOAD ? PKT_MAX_PAYLOAD : remaining);
        sx_pkt_t *p = &st->w[seq % SX_MAX_WINDOW];
        int n = pkt_encode(p->buf, PKT_MAX_SIZE, PKT_TYPE_DATA, seq, st->file + off, plen);
        p->len = n;
        st->next = seq + 1u;
        sx_timer_start(st, now_ms);
        out->buf = p->buf;
        out->len = n;
        return 1;
    }

    return 0;
}

sx_status_t sx_status(const sx_state_t *st)
{
    if (st->done)
        return SX_ST_DONE;
    if (st->gave_up)
        return SX_ST_GAVEUP;
    return SX_ST_ACTIVE;
}

long sx_timer_remaining_ms(const sx_state_t *st, long now_ms)
{
    if (!st->timer_running || st->done || st->gave_up)
        return 0;
    long rem = st->deadline_ms - now_ms;
    return rem > 0 ? rem : 0;
}

static uint8_t g_pkt[PKT_MAX_SIZE];

static int make_data(uint32_t seq, const uint8_t *payload, uint16_t len)
{
    return pkt_encode(g_pkt, sizeof(g_pkt), PKT_TYPE_DATA, seq, payload, len);
}

static int make_raw(int type, uint8_t reserved, uint32_t seq)
{
    uint8_t b[PKT_HEADER_LEN];
    memset(b, 0, sizeof(b));
    b[0] = (uint8_t)type;
    b[1] = reserved;
    b[4] = (uint8_t)(seq >> 24);
    b[5] = (uint8_t)(seq >> 16);
    b[6] = (uint8_t)(seq >> 8);
    b[7] = (uint8_t)seq;
    uint16_t c = pkt_checksum(b, sizeof(b));
    b[2] = (uint8_t)(c >> 8);
    b[3] = (uint8_t)c;
    memcpy(g_pkt, b, sizeof(b));
    return PKT_HEADER_LEN;
}

static int make_ack(uint32_t seq)
{
    return pkt_encode(g_pkt, sizeof(g_pkt), PKT_TYPE_ACK, seq, NULL, 0);
}

static int make_fin(uint32_t seq)
{
    return pkt_encode(g_pkt, sizeof(g_pkt), PKT_TYPE_FIN, seq, NULL, 0);
}

/* Clean-channel end-to-end: drives sx and rx against each other with zero loss.
 * Returns the simulated time at which the sender finished (and the receiver
 * reached DONE after its FIN linger), or -1 if it did not complete in time. */
static long run_e2e(int window, const uint8_t *file, size_t len, uint8_t *out)
{
    static sx_state_t sxs;
    static rx_state_t rxs;

    sx_init(&sxs, window, 250, file, len);
    rx_init(&rxs, 0);

    size_t written = 0;
    long t = 0;
    for (int round = 0; round < 10000; round++) {
        sx_send_t s;
        while (sx_next_send(&sxs, t, &s) == 1) {
            rx_action_t act = rx_on_packet(&rxs, t, s.buf, (size_t)s.len);
            if (act == RX_ACT_WRITE_ACK) {
                memcpy(out + written, rxs.payload, rxs.payload_len);
                written += rxs.payload_len;
            }
        }

        int n = make_ack(rxs.expected);
        sx_on_packet(&sxs, t, g_pkt, (size_t)n);

        if (sx_status(&sxs) == SX_ST_DONE) {
            if (written != len)
                return -1;
            if (rx_tick(&rxs, t + RX_LINGER_MS) != RX_ACT_DONE)
                return -1;
            return t;
        }

        long rem = sx_timer_remaining_ms(&sxs, t);
        t += (rem > 0 ? rem : 1);
        sx_tick(&sxs, t);
    }
    return -1;
}

/* xorshift32 PRNG — deterministic across platforms (32-bit ops only). */
static uint32_t g_rng;

static uint32_t rng_next(void)
{
    uint32_t x = g_rng;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    return g_rng = x;
}

typedef struct {
    uint8_t buf[PKT_MAX_SIZE];
    int len;
} chan_pkt_t;

/* Fake relay channel, mirroring cs425_relay.py's exclusive per-datagram
 * outcomes: drop 20%, else corrupt one random bit 20%, else deliver two
 * exact copies 20%, else deliver one clean copy. Returns delivery count. */
static int channel_apply(const uint8_t *buf, int len, chan_pkt_t *out)
{
    int k = (int)(rng_next() % 100u);
    if (k < 20)
        return 0;
    memcpy(out[0].buf, buf, (size_t)len);
    out[0].len = len;
    if (k < 40) {
        size_t pos = (size_t)(rng_next() % (uint32_t)len);
        out[0].buf[pos] ^= (uint8_t)(1u << (int)(rng_next() & 7u));
        return 1;
    }
    if (k < 60) {
        memcpy(out[1].buf, buf, (size_t)len);
        out[1].len = len;
        return 2;
    }
    return 1;
}

#define E2E_LOSSY_MAX_ROUNDS 200000

/* Lossy end-to-end: drives sx and rx against each other with every packet
 * (both directions) passed through channel_apply. Returns the simulated time
 * at which the transfer completed, or -1 on give-up / round cap. */
static long run_e2e_lossy(uint32_t seed, int window, const uint8_t *file, size_t len, uint8_t *out)
{
    static sx_state_t sxs;
    static rx_state_t rxs;
    static uint8_t ackbuf[PKT_MAX_SIZE];

    g_rng = (seed != 0) ? seed : 1u;
    sx_init(&sxs, window, 250, file, len);
    rx_init(&rxs, 0);

    size_t written = 0;
    long t = 0;
    for (int round = 0; round < E2E_LOSSY_MAX_ROUNDS; round++) {
        sx_send_t s;
        while (sx_next_send(&sxs, t, &s) == 1) {
            uint8_t pkt[PKT_MAX_SIZE];
            memcpy(pkt, s.buf, (size_t)s.len);
            chan_pkt_t d[2];
            int n = channel_apply(pkt, s.len, d);
            for (int i = 0; i < n; i++) {
                rx_action_t act = rx_on_packet(&rxs, t, d[i].buf, (size_t)d[i].len);
                if (act == RX_ACT_WRITE_ACK) {
                    memcpy(out + written, rxs.payload, rxs.payload_len);
                    written += rxs.payload_len;
                }
                if (act == RX_ACT_WRITE_ACK || act == RX_ACT_ACK) {
                    int alen = pkt_encode(ackbuf, sizeof(ackbuf), PKT_TYPE_ACK, rxs.expected, NULL, 0);
                    chan_pkt_t a[2];
                    int m = channel_apply(ackbuf, alen, a);
                    for (int j = 0; j < m; j++)
                        sx_on_packet(&sxs, t, a[j].buf, (size_t)a[j].len);
                }
            }
        }

        if (rx_tick(&rxs, t) == RX_ACT_GIVEUP)
            return -1;
        if (sx_status(&sxs) == SX_ST_GAVEUP)
            return -1;
        if (sx_status(&sxs) == SX_ST_DONE) {
            if (written != len)
                return -1;
            if (rx_tick(&rxs, t + RX_LINGER_MS) != RX_ACT_DONE)
                return -1;
            return t;
        }

        long rem = sx_timer_remaining_ms(&sxs, t);
        t += (rem > 0 ? rem : 1);
        sx_tick(&sxs, t);
    }
    return -1;
}
